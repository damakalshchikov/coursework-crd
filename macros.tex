% Секция без нумерации, которая H1 и подчёркивание
% содержит в верхнем колонтитуле название
\newcommand{\unnsection}[1]{
    \clearpage
    \section*{#1}
    \addcontentsline{toc}{section}{#1}
    \markboth{#1}{#1}
    \thispagestyle{plain}
}

% Подсекция без нумерации
\newcommand{\unnsubsection}[1]{
    \subsection*{#1}
    \addcontentsline{toc}{subsection}{#1}
}

% Подподсекция без нумерации
\newcommand{\unnsubsubsection}[1]{
    \subsubsection*{#1}
    \addcontentsline{toc}{subsubsection}{#1}
}
